1. 现状与挑战

agent-service 当前零测试、零测试依赖、零测试配置。这不是补几个测试文件能解决的——需要一套完整的测试基础设施。

1.1 当前状态

测试文件

零 无 tests/ 目录、无 conftest.py、无 fixture/mock 代码

测试依赖

零 pyproject.toml 无 pytest、无 pytest-asyncio、无 httpx test 依赖

测试配置

零 无 [tool.pytest.ini_options]、无 ruff/mypy 配置段

1.2 核心挑战：模块级副作用

agent-service 的代码在导入即执行模块级别实例化了多个外部依赖，这是测试隔离的最大障碍：

模块模块级副作用测试影响
core/db.pycreate_engine(settings.database_url) + SessionLocal导入即连 PostgreSQL，CI 环境无数据库直接报错
tools/registry.pyredis.Redis.from_url(settings.redis_url)导入即连 Redis，无 Redis 时模块加载失败
services/session_service.pyredis_client + _id_seq = itertools.count(1)Redis 连接 + ID 序列器全局状态，测试间状态泄漏
services/memory_service.pyredis_client同上，记忆系统测试与主服务共享 Redis 连接
main.pylifespan 启动 NacosRegistrarTestClient 触发 lifespan → 尝试 Nacos 注册
核心矛盾：不能简单地 import agent.tools.registry 来测试工具函数——导入瞬间就尝试连接 Redis 和 PostgreSQL。必须用环境变量注入 + mock 替换的方式在导入前拦截这些副作用，否则任何测试都无法启动。

1.3 外部 I/O 依赖图谱

Agent 的每次 run_turn 涉及 4 类外部 I/O，测试必须全部拦截：

依赖类型涉及模块拦截方式
PostgreSQLcore/db.py → 全部 ServiceSQLite in-memory + create_all()
Redisregistry.py, session_service.py, memory_service.pyfakeredis 或 monkeypatch 替换 redis_client
httpx → 微服务registry.py 12 个工具函数, session_service.run_turnrespx 拦截 httpx 请求 / monkeypatch
LLM APIplanner.py llm_plan() → OpenAI 兼容接口mock httpx.post / 预录制响应
Nacosmain.py lifespan跳过 lifespan / mock NacosRegistrar

2. 测试框架总体架构

采用五层测试金字塔，从确定性单元测试到非确定性 Agent 行为评测，每层有明确的工具、目标和量化指标。

2.1 各层定位与工具选型

层测试什么工具确定性量化指标运行时机
L1 单元测试planner 意图解析、classify_risk 阈值、memory 三关判据、工具函数参数构造pytest + pytest-asyncio + fakeredis + respx100% 确定line coverage ≥90%, branch ≥85%每次提交
L2 集成测试21 个 API 端点、SSE 流式、确认流程、Skill CRUD、记忆 CRUDhttpx.AsyncClient + ASGITransport100% 确定端点覆盖率 100%, 确认流程通过率 100%每次提交
L3 行为评测工具选择准确率、参数正确率、规划遵循度、幻觉检测、确认合规DeepEval + 自定义 metric非确定（LLM 裁判）工具准确率 ≥85%, Faithfulness ≥0.9每日/每版
L4 端到端完整 run_turn 流程、Canvas Norm 依赖图走路、异步 ack 协议pytest + LangGraph ainvoke + state-diff半确定任务完成率 ≥75%, pass³ ≥0.6每次合并
L5 CI 回归提示词回归、模型升级对比、安全红队promptfoo --ci半确定回归通过率 ≥90%, 越狱率 ≤5%每次 PR

3.2 conftest.py：副作用拦截基础设施

conftest.py 是整个测试框架的核心——它在任何 import agent.* 之前拦截模块级副作用：

# tests/conftest.py
"""全局测试夹具：在导入 agent 模块前拦截所有模块级副作用。"""
import os
import sys

# ── 第一步：环境变量注入（必须在 import agent.* 之前）──
os.environ["VIBEPAPER_DATABASE_URL"] = "sqlite:///test_agent.db"
os.environ["VIBEPAPER_REDIS_URL"] = "redis://fake:6379/0"
os.environ["VIBEPAPER_LLM_API_KEY"] = "test-key"
os.environ["VIBEPAPER_LLM_BASE_URL"] = "http://mock-llm:8080/v1"

# ── 第二步：导入 fakeredis 并 patch redis 模块 ──
import fakeredis
import redis
# 创建一个共享的 fake redis server，所有 redis.Redis() 调用都指向它
_fake_server = fakeredis.FakeServer()
_original_redis_from_url = redis.Redis.from_url
def _patched_from_url(cls, *args, **kwargs):
    return fakeredis.FakeStrictRedis(server=_fake_server)
redis.Redis.from_url = classmethod(_patched_from_url)

# ── 第三步：现在可以安全导入 agent 模块 ──
from agent.core.db import Base, engine, SessionLocal
from agent.core.config import get_settings

# ── 第四步：用 SQLite 替换 PostgreSQL engine ──
from sqlalchemy import create_engine as _create_engine
from sqlalchemy.pool import StaticPool
_test_engine = _create_engine(
    "sqlite://",
    connect_args={"check_same_thread": False},
    poolclass=StaticPool,
)
# 替换全局 engine 和 SessionLocal
import agent.core.db as _db
_db.engine = _test_engine
_db.SessionLocal.configure(bind=_test_engine)

import pytest
from sqlalchemy.orm import Session

@pytest.fixture
def db_session() -> Session:
    """每个测试用例获得干净的内存数据库。"""
    Base.metadata.create_all(_test_engine)
    session = SessionLocal()
    try:
        yield session
    finally:
        session.close()
        Base.metadata.drop_all(_test_engine)

@pytest.fixture
def fake_redis():
    """共享的 fakeredis 实例，每个测试前 flush。"""
    r = fakeredis.FakeStrictRedis(server=_fake_server)
    r.flushall()
    return r

@pytest.fixture
def mock_canvas_response():
    """预制的 canvas-service 响应。"""
    return {
        "id": 1, "name": "测试画布", "version": 1,
        "nodes": [
            {"id": 1, "type": "text", "x": 100, "y": 100, "params": {"prompt": "脚本"}},
            {"id": 2, "type": "image", "x": 200, "y": 150, "params": {"prompt": "角色"}},
        ],
        "edges": [{"sourceNodeId": 1, "targetNodeId": 2}],
    }
设计要点：conftest.py 的执行顺序是环境变量 → patch redis → 导入 agent → 替换 engine。这个顺序不能打乱——redis patch 必须在 import agent.tools.registry 之前，否则 registry 模块顶层的 redis.Redis.from_url() 已经用真实 URL 创建了连接。

3.4 单元测试示例：planner 意图解析

planner 的 plan() 是纯函数（不依赖外部服务），是单元测试的理想目标：

# tests/unit/test_planner.py
"""测试规划器的意图解析和动作生成。"""
import pytest
from agent.agent.planner import plan, PlannedAction, INTENT_RULES

class TestIntentParsing:
    """意图解析：8 种意图 × 正则匹配。"""

    @pytest.mark.parametrize("content,expected_intent", [
        ("创建一个图片节点", "create"),
        ("新建一个视频", "create"),
        ("make a text node", "create"),
        ("连接节点1和节点2", "connect"),
        ("删除选中节点", "delete"),
        ("整理画布布局", "layout"),
        ("修改参数为高清", "update"),
        ("搜索海洋素材", "search"),
        ("换成 qwen 模型", "model"),
        ("生成一张图片", "generate"),
    ])
    def test_intent_detection(self, content, expected_intent):
        """每种意图关键词应被正确识别。"""
        actions = plan(content)
        assert len(actions) > 0
        # 验证意图对应的工具被调用
        if expected_intent == "create":
            assert any(a.tool_name == "create_nodes" for a in actions)
        elif expected_intent == "delete":
            assert any(a.tool_name == "delete_nodes" for a in actions)
        elif expected_intent == "search":
            assert any(a.tool_name == "search_assets" for a in actions)

    def test_create_image_node(self):
        """'创建图片'应生成 image 类型节点。"""
        actions = plan("创建一张海洋图片")
        create_action = next(a for a in actions if a.tool_name == "create_nodes")
        assert len(create_action.params["nodes"]) >= 1
        assert create_action.params["nodes"][0]["type"] == "image"

    def test_create_multiple_types(self):
        """'创建图片和视频'应生成两种节点。"""
        actions = plan("创建一张图片和一段视频")
        create_action = next(a for a in actions if a.tool_name == "create_nodes")
        types = [n["type"] for n in create_action.params["nodes"]]
        assert "image" in types
        assert "video" in types

    def test_generate_triggers_submit(self):
        """'生成'意图应同时触发 create_nodes + submit_generation。"""
        actions = plan("生成一张图片")
        tool_names = [a.tool_name for a in actions]
        assert "create_nodes" in tool_names
        assert "submit_generation" in tool_names

    def test_empty_canvas_fallback(self):
        """无法识别意图时回退到 get_canvas_summary。"""
        actions = plan("???")
        assert len(actions) == 1
        assert actions[0].tool_name == "get_canvas_summary"

3.5 单元测试示例：风险分级

# tests/unit/test_registry.py
"""测试工具白名单和风险分级阈值。"""
import pytest
from agent.tools.registry import classify_risk, TOOLS

class TestRiskClassification:
    """PRD §5.2.1 风险分级规则。"""

    def test_read_tools_no_risk(self):
        """read 类工具永远无需确认。"""
        for tool_name in ["get_canvas_summary", "list_models", "search_assets"]:
            risk, reason = classify_risk(tool_name, {})
            assert risk == "read"
            assert reason is None

    def test_delete_always_high(self):
        """删除操作恒定 high risk。"""
        risk, reason = classify_risk("delete_nodes", {"node_ids": [1]})
        assert risk == "high"

    def test_batch_create_over_20_escalates(self):
        """批量创建 >20 升级为 high。"""
        nodes = [{"type": "text"}] * 21
        risk, reason = classify_risk("create_nodes", {"nodes": nodes})
        assert risk == "high"
        assert "批量" in reason or "20" in reason

    def test_batch_create_under_20_is_low(self):
        """批量创建 ≤20 保持 low。"""
        nodes = [{"type": "text"}] * 20
        risk, reason = classify_risk("create_nodes", {"nodes": nodes})
        assert risk == "low"

    def test_submit_generation_always_high(self):
        """submit_generation 恒定 high（涉及计费）。"""
        risk, _ = classify_risk("submit_generation", {"estimated_cost": 10})
        assert risk == "high"

    def test_all_tools_registered(self):
        """确保 12 个工具全部注册。"""
        expected = {
            "get_canvas_summary", "get_selected_nodes", "list_models",
            "search_assets", "create_nodes", "connect_nodes",
            "layout_nodes", "update_node_config", "delete_nodes",
            "change_model", "replace_output", "submit_generation",
        }
        assert set(TOOLS.keys()) == expected

5. Agent 行为评测层（L3）

这一层测试的不是"代码对不对"，而是"Agent 聪不聪明"。用 DeepEval 框架将 LLM 评测融入 pytest 工作流，对工具选择、规划质量、幻觉、确认合规做量化评分。

5.1 评测框架选型：DeepEval

选择 DeepEval 而非 LangSmith / Langfuse，原因如下：

维度DeepEvalLangSmithpromptfoo
集成方式pytest 原生（@assert_test）SDK 远程调用YAML 声明式
Agent 专属指标7 项开箱即用需自定义基础断言
CI 门控阈值拦截，原生支持需额外配置--ci 原生
本地运行取决于裁判 LLM云端100% 本地
本框架定位L3 行为评测主框架生产监控（补充）L5 CI 回归

5.2 七项 Agent 核心指标

指标定义计算方式目标阈值
工具选择准确率选择正确工具的次数 / 总工具调用次数逐条对比 expected_tool vs actual_tool≥85%
参数正确率工具参数提取正确的比例逐字段 JSON diff≥80%
执行成功率工具调用后环境状态达预期变化的比例state-diff 验证≥90%

5.4 工具选择准确率评测实现

# tests/eval/test_tool_selection.py
"""工具选择准确率评测——量化 Agent 选对工具的能力。"""
import json
import pytest
from pathlib import Path
from agent.agent.planner import plan

DATASET = Path(__file__).parent / "datasets" / "golden_cases.json"

def load_cases():
    return json.loads(DATASET.read_text(encoding="utf-8"))

@pytest.mark.eval
class TestToolSelectionAccuracy:

    @pytest.mark.parametrize("case", load_cases())
    def test_primary_tool_correct(self, case):
        """主工具选择是否正确。"""
        actions = plan(
            case["instruction"],
            canvas=case.get("canvas_context"),
            selected_nodes=case.get("selected_nodes"),
        )
        actual_tools = [a.tool_name for a in actions]
        expected_primary = case["expected"]["primary_tool"]
        assert expected_primary in actual_tools, (
            f"期望主工具 {expected_primary}，实际 {actual_tools}"
        )

    @pytest.mark.parametrize("case", load_cases())
    def test_tool_sequence_correct(self, case):
        """工具序列是否包含所有期望工具。"""
        actions = plan(
            case["instruction"],
            canvas=case.get("canvas_context"),
            selected_nodes=case.get("selected_nodes"),
        )
        actual_tools = {a.tool_name for a in actions}
        expected_tools = set(case["expected"]["tools"])
        missing = expected_tools - actual_tools
        assert not missing, f"缺失工具: {missing}"

    @pytest.mark.parametrize("case", load_cases())
    def test_risk_level_correct(self, case):
        """风险等级判定是否正确。"""
        from agent.tools.registry import classify_risk
        actions = plan(
            case["instruction"],
            canvas=case.get("canvas_context"),
            selected_nodes=case.get("selected_nodes"),
        )
        expected_risk = case["expected"]["risk_level"]
        for action in actions:
            risk, _ = classify_risk(action.tool_name, action.params)
            if risk == "high":
                assert case["expected"]["requires_confirm"], (
                    f"{action.tool_name} 期望 high 但用例未标记需确认"
                )

    def test_tool_selection_accuracy_rate(self):
        """整体工具选择准确率（量化指标）。"""
        cases = load_cases()
        correct = 0
        total = len(cases)
        for case in cases:
            actions = plan(case["instruction"])
            actual = {a.tool_name for a in actions}
            if case["expected"]["primary_tool"] in actual:
                correct += 1
        accuracy = correct / total
        print(f"\n工具选择准确率: {correct}/{total} = {accuracy:.1%}")
        assert accuracy >= 0.85, f"准确率 {accuracy:.1%} 低于阈值 85%"

5.5 确认机制合规评测

# tests/eval/test_confirm_compliance.py
"""确认机制合规评测——high risk 操作必须 100% 触发确认。"""
import json
import pytest
from pathlib import Path
from agent.agent.planner import plan
from agent.tools.registry import classify_risk

DATASET = Path(__file__).parent / "datasets" / "golden_cases.json"

@pytest.mark.eval
class TestConfirmCompliance:

    def test_all_high_risk_actions_require_confirm(self):
        """所有 high risk 操作必须标记为需确认。"""
        cases = json.loads(DATASET.read_text(encoding="utf-8"))
        violations = []
        for case in cases:
            if not case["expected"].get("requires_confirm"):
                continue
            actions = plan(case["instruction"],
                          canvas=case.get("canvas_context"),
                          selected_nodes=case.get("selected_nodes"))
            for action in actions:
                risk, reason = classify_risk(action.tool_name, action.params)
                if risk == "high" and not reason:
                    violations.append(
                        f"TC={case['id']} tool={action.tool_name} "
                        f"risk=high 但无确认原因")
        assert not violations, (
            f"确认合规违规 {len(violations)} 条:\n" + "\n".join(violations))

    def test_confirm_compliance_rate(self):
        """确认合规率量化指标。"""
        cases = json.loads(DATASET.read_text(encoding="utf-8"))
        high_risk_total = 0
        confirmed = 0
        for case in cases:
            actions = plan(case["instruction"])
            for action in actions:
                risk, reason = classify_risk(action.tool_name, action.params)
                if risk == "high":
                    high_risk_total += 1
                    if reason:
                        confirmed += 1
        rate = confirmed / high_risk_total if high_risk_total else 1.0
        print(f"\n确认合规率: {confirmed}/{high_risk_total} = {rate:.1%}")
        assert rate == 1.0, "确认合规率必须 100%"

6.2 pass^k 可靠性测试

单次成功可能是运气——pass^k 要求同一任务连续 k 次全部成功才算通过[3]。这是 Agent 真实可靠性的核心指标：

# tests/e2e/test_reliability.py
"""pass^k 可靠性测试——同一任务重复执行 k 次验证稳定性。"""
import pytest
from agent.agent.planner import plan

@pytest.mark.e2e
class TestReliability:

    @pytest.mark.parametrize("execution", range(3))
    def test_plan_stability_pass3(self, execution):
        """pass³: 同一指令执行 3 次结果必须一致（规则引擎）。"""
        actions = plan("创建一张赛博朋克图片")
        tool_names = [a.tool_name for a in actions]
        # 规则引擎是确定性的，3 次结果必须完全一致
        assert "create_nodes" in tool_names
        assert "submit_generation" in tool_names
        # 记录每次结果用于统计
        # 实际 LLM 规划的 pass^k 需在 eval 层用 temperature=0 跑

    def test_pass_k_metric(self):
        """计算 pass^k 量化指标。"""
        import json
        from pathlib import Path
        cases = json.loads(
            (Path(__file__).parent.parent / "eval" / "datasets"
             / "golden_cases.json").read_text(encoding="utf-8"))
        k = 3  # pass³
        all_pass = 0
        for case in cases:
            passed_all = True
            for _ in range(k):
                actions = plan(case["instruction"])
                actual = {a.tool_name for a in actions}
                if case["expected"]["primary_tool"] not in actual:
                    passed_all = False
                    break
            if passed_all:
                all_pass += 1
        pass_k = all_pass / len(cases)
        print(f"\npass^{k} = {pass_k:.1%} ({all_pass}/{len(cases)})")
        assert pass_k >= 0.6, f"pass^{k} {pass_k:.1%} 低于阈值 60%"

# tests/metrics_collector.py
"""CI 运行后采集所有量化指标，输出 JSON 仪表盘。"""
import json
import subprocess
from pathlib import Path

def collect_coverage():
    """从 pytest-cov XML 提取覆盖率。"""
    xml = Path("coverage.xml").read_text()
    # 解析 line-rate 和 branch-rate
    import xml.etree.ElementTree as ET
    root = ET.fromstring(xml)
    return {
        "line_coverage": float(root.attrib["line-rate"]) * 100,
        "branch_coverage": float(root.attrib["branch-rate"]) * 100,
    }

def collect_tool_accuracy():
    """运行工具选择准确率测试并采集结果。"""
    result = subprocess.run(
        ["pytest", "tests/eval/test_tool_selection.py::TestToolSelectionAccuracy::test_tool_selection_accuracy_rate",
         "-v", "--tb=short", "-s"],
        capture_output=True, text=True)
    # 从输出解析准确率
    for line in result.stdout.split("\n"):
        if "工具选择准确率" in line:
            return {"tool_accuracy": line.strip()}
    return {"tool_accuracy": "unknown"}

def generate_dashboard():
    """生成指标仪表盘 JSON。"""
    dashboard = {
        "timestamp": "2026-08-07T06:00:00Z",
        "metrics": {
            **collect_coverage(),
            **collect_tool_accuracy(),
        },
        "thresholds": {
            "line_coverage": 90,
            "branch_coverage": 85,
            "tool_accuracy": 85,
        },
        "passed": True,  # 与阈值对比后设置
    }
    Path("test-metrics.json").write_text(
        json.dumps(dashboard, indent=2, ensure_ascii=False))
    return dashboard

if __name__ == "__main__":
    generate_dashboard()
